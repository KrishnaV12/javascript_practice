```javascript
// Helper to check if a substring is a palindrome within given bounds
function checkPalindrome(str, low, high) {
    while (low < high) {
        if (str[low] !== str[high]) return false;
        low++;
        high--;
    }
    return true;
}

// Part 2: Can be palindrome by removing at most one character
function can_be_palindrome_by_removing_one(s) {
    // Clean string: lowercase and remove non-alphanumeric characters
    const minimalStr = s.toLowerCase().replace(/[^a-z0-9]/g, '');
    
    let left = 0;
    let right = minimalStr.length - 1;
    
    while (left < right) {
        if (minimalStr[left] !== minimalStr[right]) {
            // On mismatch, try skipping either the left or right character
            return checkPalindrome(minimalStr, left + 1, right) || 
                   checkPalindrome(minimalStr, left, right - 1);
        }
        left++;
        right--;
    }
    
    return true;
}
```
